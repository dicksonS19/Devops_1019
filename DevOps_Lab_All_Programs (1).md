## Exp 1 – Git Essentials (Commit, Push, Merge, Pull Request)

```bash
git config --global user.name "Your Name"
git config --global user.email "you@example.com"

mkdir myproject && cd myproject
git init
touch text1.txt
git add .
git status
git commit -m "Initial commit"
git log
git log --oneline

# rename branch to main
git branch -M main

# connect to GitHub (create an empty repo on GitHub first)
git remote add origin https://github.com/<username>/Exp_2.git
git push -u origin main

# --- Branch, merge and pull request ---
git checkout -b feature-1
echo "new feature" > feature.txt
git add .
git commit -m "Add feature"
git push -u origin feature-1
# Open a Pull Request on GitHub (feature-1 -> main) and merge it

# Merge locally instead
git checkout main
git merge feature-1

# Pull latest changes
git pull origin main
```

## Exp 2 & 3 – Flask App Containerization and Docker Hub Deployment

### app.py
```python
from flask import Flask

app = Flask(__name__)

@app.route("/")
def home():
    return """
    <html>
    <head>
        <title>Flask Docker App</title>
        <style>
            body {
                margin: 0;
                height: 100vh;
                display: flex;
                justify-content: center;
                align-items: center;
                font-family: Arial, sans-serif;
                background: linear-gradient(135deg, #00c6ff, #0072ff);
                color: white;
                text-align: center;
            }
            .card {
                background: rgba(255,255,255,0.15);
                padding: 40px 60px;
                border-radius: 15px;
            }
        </style>
    </head>
    <body>
        <div class="card">
            <h1>🚀 Welcome to Flask + Docker!</h1>
            <p>This page is running inside a Docker container.</p>
        </div>
    </body>
    </html>
    """

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
```

### requirements.txt
```
flask
```

### Dockerfile
```dockerfile
FROM python:3.9-slim

WORKDIR /app

COPY requirements.txt .
RUN pip install -r requirements.txt

COPY . .

EXPOSE 5000

CMD ["python", "app.py"]
```

### Commands
```bash

docker run -p 5050:5000 flask-docker-app
```

## Exp 4 – Jenkins Deployment using Docker Compose + Static Site Pipeline

### docker-compose.yml
```yaml
services:
  jenkins:
    image: jenkins/jenkins:lts
    container_name: jenkins
    restart: unless-stopped
    ports:
      - "8080:8080"
      - "50000:50000"
    volumes:
      - jenkins_home:/var/jenkins_home

volumes:
  jenkins_home:
```

### Commands
```bash
mkdir ~/jenkins-static-site
cd ~/jenkins-static-site
docker compose up -d
docker ps
docker exec jenkins cat /var/jenkins_home/secrets/initialAdminPassword
docker logs jenkins
docker compose restart
docker compose down
```
Then open `http://localhost:8080`, install suggested plugins, and install the **HTML Publisher** plugin (Manage Jenkins → Plugins).

### Pipeline script (job name: `karunya-site-pipeline`)
```groovy
pipeline {
    agent any

    stages {
        stage('Generate site') {
            steps {
                sh '''
                mkdir -p site

                cat > site/index.html << 'EOF'
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>Karunya University — Simple Site</title>
  <link rel="stylesheet" href="style.css">
</head>
<body>
  <header>
    <h1>Karunya University</h1>
    <p>Values — Knowledge — Service</p>
  </header>
  <main>
    <section class="card">
      <h2>About Karunya University</h2>
      <p>Karunya Institute of Technology and Sciences is a leading centre for higher education in India.</p>
    </section>
  </main>
  <footer>
    <p>&copy; 2026 Karunya University — Generated by Jenkins</p>
  </footer>
</body>
</html>
EOF

                cat > site/style.css << 'EOF'
body { margin: 0; font-family: Arial, sans-serif; background: #f2f2f2; }
header { background: #0d3b9e; color: white; padding: 40px 20%; }
header h1 { margin: 0; }
main { padding: 20px 20%; }
.card { background: white; padding: 15px 20px; box-shadow: 0 1px 4px rgba(0,0,0,0.2); }
footer { text-align: center; color: #777; font-size: 12px; padding: 40px; }
EOF
                '''
            }
        }

        stage('Publish site') {
            steps {
                publishHTML(target: [
                    allowMissing: false,
                    alwaysLinkToLastBuild: true,
                    keepAll: true,
                    reportDir: 'site',
                    reportFiles: 'index.html',
                    reportName: 'Karunya University - Simple Site'
                ])
            }
        }
    }

    post {
        always {
            archiveArtifacts artifacts: 'site/**', fingerprint: true
        }
    }
}
```

### Commands
```bash
docker --version
choco install kubernetes-cli
kubectl version --client
choco install minikube
minikube start --driver=docker
kubectl get nodes

# Pod
kubectl apply -f nginx-pod.yaml
kubectl get pods
kubectl describe pod nginx-pod
kubectl port-forward pod/nginx-pod 8080:80      # open http://localhost:8080
kubectl delete pod nginx-pod

# Deployment, scaling, rolling update, rollback
kubectl create deployment my-nginx --image=nginx
kubectl scale deployment my-nginx --replicas=3
kubectl get pods
kubectl set image deployment/my-nginx nginx=nginx:1.25
kubectl rollout status deployment/my-nginx
kubectl rollout undo deployment/my-nginx

# Service
kubectl expose deployment my-nginx --type=NodePort --port=80
minikube service my-nginx --url
kubectl delete svc my-nginx

# Cleanup
kubectl delete deployment my-nginx
kubectl delete all --all -n default
minikube stop
minikube delete
```

## Exp 9 – Deploy a Python Flask Web App Locally using Ansible

### Prerequisites
```bash
sudo apt update
sudo apt upgrade -y
sudo apt install -y python3 python3-pip python3-venv git curl
python3 --version
pip3 --version
sudo apt-add-repository --yes --update ppa:ansible/ansible
sudo apt install -y ansible
ansible --version
```

### Project layout
```
flask-app/
├── inventory.ini
├── playbook.yml
└── app/
    └── app.py
```

### inventory.ini
```ini
[local]
localhost ansible_connection=local
```

### app/app.py
```python
from flask import Flask, jsonify

app = Flask(__name__)

@app.route("/")
def home():
    return jsonify(msg="Hello from Ansible-deployed Flask!", status="ok")

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
```

### playbook.yml
```yaml
- name: Deploy Flask app to localhost with systemd
  hosts: local
  become: true
  vars:
    app_dir: /opt/ansible_flask_app
    venv_dir: /opt/ansible_flask_app/venv

  pre_tasks:
    - name: Ensure Python and system tools present (apt)
      ansible.builtin.apt:
        name:
          - python3
          - python3-pip
          - python3-venv
        state: present
        update_cache: true

  tasks:
    - name: Create application directory
      ansible.builtin.file:
        path: "{{ app_dir }}"
        state: directory
        mode: "0755"

    - name: Copy application source
      ansible.builtin.copy:
        src: app/
        dest: "{{ app_dir }}/"
        mode: "0644"
      notify: restart app

    - name: Create venv
      ansible.builtin.command:
        cmd: python3 -m venv {{ venv_dir }}
        creates: "{{ venv_dir }}/bin/activate"

    - name: Upgrade pip
      ansible.builtin.pip:
        name: pip
        state: latest
        virtualenv: "{{ venv_dir }}"

    - name: Install app dependencies (flask, gunicorn)
      ansible.builtin.pip:
        name:
          - flask
          - gunicorn
        virtualenv: "{{ venv_dir }}"

    - name: Render wsgi entrypoint
      ansible.builtin.copy:
        dest: "{{ app_dir }}/wsgi.py"
        content: |
          from app import app

          if __name__ == "__main__":
              app.run()
        mode: "0644"
      notify: restart app

    - name: Place systemd unit
      ansible.builtin.copy:
        dest: /etc/systemd/system/ansible-flask.service
        content: |
          [Unit]
          Description=Gunicorn for Flask app
          After=network.target

          [Service]
          WorkingDirectory={{ app_dir }}
          ExecStart={{ venv_dir }}/bin/gunicorn --workers 2 --bind 127.0.0.1:8000 wsgi:app
          Restart=always

          [Install]
          WantedBy=multi-user.target
        mode: "0644"
      notify:
        - daemon-reload
        - restart app

    - name: Enable service on boot
      ansible.builtin.systemd:
        name: ansible-flask
        enabled: true

    - name: Start service now
      ansible.builtin.systemd:
        name: ansible-flask
        state: started

  handlers:
    - name: daemon-reload
      ansible.builtin.systemd:
        daemon_reload: true

    - name: restart app
      ansible.builtin.systemd:
        name: ansible-flask
        state: restarted
```
> The playbook imports `app` from `app.py`; `wsgi:app` works because `wsgi.py` does `from app import app`.

### Run and verify
```bash
ansible -i inventory.ini all -m ping
ansible-playbook playbook.yml -i inventory.ini -K
# run again to show idempotency
ansible-playbook playbook.yml -i inventory.ini -K

sudo systemctl status ansible-flask.service
sudo journalctl -u ansible-flask.service -f
curl http://127.0.0.1:8000
```

### requirements.yml

```yaml
  - name: community.docker
```

```bash
ansible-galaxy collection install -r requirements.yml
```

### inventory.ini
```ini
[local]
127.0.0.1 ansible_connection=local
```

### docker_k8s.yml (sample playbook from the manual)
```yaml
- name: Docker and Kubernetes Deployment
  hosts: local
  tasks:
    - name: Run Docker Container
      community.docker.docker_container:
        name: mynginx
        image: nginx
        state: started
        ports:
          - "8080:80"

    - name: Deploy to Kubernetes
      kubernetes.core.k8s:
        state: present
        definition:
          apiVersion: v1
          kind: Pod
          metadata:
            name: nginx-pod
          spec:
            containers:
              - name: nginx
                image: nginx
                ports:
                  - containerPort: 80
```

### docker.yml (Docker part – NGINX on port 8080)
```yaml
- name: docker.yml
  hosts: local
  tasks:
    - name: Ensure Docker is running
      ansible.builtin.service:
        name: docker
        state: started
        enabled: true
      become: true

    - name: Run NGINX container on 8080
      community.docker.docker_container:
        name: web
        image: nginx:alpine
        state: started
        ports:
          - "8080:80"

    - name: Show URL
      ansible.builtin.debug:
        msg: "Open http://localhost:8080"
```

### k8s-nginx.yaml (Deployment + Service for the kind cluster)
```yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nginx
spec:
  replicas: 1
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
        - name: nginx
          image: nginx:alpine
          ports:
            - containerPort: 80
---
apiVersion: v1
kind: Service
metadata:
  name: nginx
spec:
  type: NodePort
  selector:
    app: nginx
  ports:
    - port: 80
      targetPort: 80
```

### k8s.yml (Kubernetes part using kind)
```yaml
- name: k8s.yml
  hosts: local
  tasks:
    - name: Install kubectl
      ansible.builtin.shell: |
        curl -LO https://dl.k8s.io/release/$(curl -Ls https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl
        chmod +x kubectl && sudo mv kubectl /usr/local/bin/kubectl
      args:
        creates: /usr/local/bin/kubectl

    - name: Install kind
      ansible.builtin.shell: |
        curl -Lo kind https://kind.sigs.k8s.io/dl/v0.23.0/kind-linux-amd64
        chmod +x kind && sudo mv kind /usr/local/bin/kind
      args:
        creates: /usr/local/bin/kind

    - name: Create kind cluster if missing
      ansible.builtin.shell: |
        kind get clusters | grep -q '^simple$' || kind create cluster --name simple
      register: kind_out
      changed_when: "'Creating cluster' in kind_out.stdout"

    - name: Apply NGINX Deployment & Service
      ansible.builtin.command: kubectl apply -f k8s-nginx.yaml
      register: apply_out
      changed_when: "'created' in apply_out.stdout or 'configured' in apply_out.stdout"

    - name: Show services
      ansible.builtin.command: kubectl get svc
      register: svc
      changed_when: false

    - name: Print services
      ansible.builtin.debug:
        var: svc.stdout_lines
```
